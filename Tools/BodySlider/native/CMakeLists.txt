cmake_minimum_required(VERSION 3.20)
project(BodySlider LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "BodySlider requires MSVC with -A Win32 for the stock x86 client")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(BodySlider SHARED BodySlider.cpp BodySlider.def)
target_compile_definitions(BodySlider PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySlider PRIVATE /W4 /EHsc /permissive-)
add_executable(BodySliderMathTest test_shape.cpp)
target_compile_options(BodySliderMathTest PRIVATE /W4 /UNDEBUG)
enable_testing()
add_test(NAME shape-math COMMAND BodySliderMathTest)
add_executable(BodySliderNativeTest test_native.cpp)
target_compile_definitions(BodySliderNativeTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySliderNativeTest PRIVATE /W4 /EHsc /UNDEBUG)
target_link_options(BodySliderNativeTest PRIVATE /BASE:0x10000000 /DYNAMICBASE:NO /FIXED)
add_test(NAME native-integration COMMAND BodySliderNativeTest)
add_executable(BodySliderDeformTest test_deform.cpp)
target_compile_options(BodySliderDeformTest PRIVATE /W4 /UNDEBUG)
add_test(NAME deformation-math COMMAND BodySliderDeformTest)
add_executable(BodySliderMeshRuntimeTest test_mesh_runtime.cpp)
target_compile_definitions(BodySliderMeshRuntimeTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySliderMeshRuntimeTest PRIVATE /W4 /EHsc /UNDEBUG)
add_test(NAME mesh-runtime COMMAND BodySliderMeshRuntimeTest)
