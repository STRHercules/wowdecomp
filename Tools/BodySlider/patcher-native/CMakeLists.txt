cmake_minimum_required(VERSION 3.20)
project(BodySliderPatch LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "BodySliderPatch requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(BodySliderPatch WIN32 BodySliderPatch.cpp)
target_compile_definitions(BodySliderPatch PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BodySliderPatch PRIVATE /W4 /EHsc /Os /Gy)
target_link_options(BodySliderPatch PRIVATE /OPT:REF /OPT:ICF)
target_link_libraries(BodySliderPatch PRIVATE advapi32 comdlg32 shell32 user32)
